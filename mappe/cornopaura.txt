<script>
(function () {
  function initMapCornoPaura() {
    if (!window.L) return;
    var mapEl = document.getElementById('rr-map-cornopaura');
    if (!mapEl) return;

    var map = L.map(mapEl).setView([45.79, 10.91], 11);
    L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', { maxZoom:19, attribution:'&copy; OpenStreetMap contributors' }).addTo(map);

    var gConfine = L.layerGroup().addTo(map);
    if (window.omnivore) {
      var confine = omnivore.kml('/kml/S0336_confine.kml');
      confine.on('ready', function () {
        gConfine.addLayer(confine);
        var b = confine.getBounds();
        if (b && b.isValid()) map.fitBounds(b.pad(0.08), { maxZoom: 14 });
      });
    }

    var cb = document.getElementById('rr-cb-cornopaura');
    if (cb) cb.addEventListener('change', function () { cb.checked ? map.addLayer(gConfine) : map.removeLayer(gConfine); });

    setTimeout(function () { map.invalidateSize(); }, 250);
    setTimeout(function () { map.invalidateSize(); }, 800);
  }

  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', initMapCornoPaura);
  else initMapCornoPaura();
})();
</script>
